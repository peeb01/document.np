Format: np
# Define typed struct
struct User:
    string name
    int age

# Function returning typed value
fn create_user(string n, int a) -> User:
    return User(n, a)

User u = create_user("Alice", 25)
print("Welcome:", u.name)

# List comprehension
array doubled = [x * 2 for x in range(1, 4)]
print("Result:", doubled)
